import { Component, computed, inject } from '@angular/core'; import { Router, RouterLink } from '@angular/router'; import { AuthService } from './auth.service'; import { OrderService } from './order.service'; import { OrderStatus } from './order.model'; @Component({ selector: 'app-restaurant-dashboard', standalone: true, imports: [RouterLink], templateUrl: './restaurant-dashboard.component.html', styleUrl: './restaurant-dashboard.component.css', }) export class RestaurantDashboardComponent { private readonly orderService = inject(OrderService); private readonly auth = inject(AuthService); private readonly router = inject(Router); readonly orders = this.orderService.orders; readonly totalOrders = computed(() => this.orders().length); readonly activeOrders = computed( () => this.orders().filter(order => order.status !== 'Entregado').length, ); readonly completedOrders = computed( () => this.orders().filter(order => order.status === 'Entregado').length, ); readonly statuses: OrderStatus[] = [ 'Recibido', 'En preparación', 'Listo para recoger', 'En camino', 'Entregado', ]; changeStatus(orderId: string, event: Event): void { const selectedStatus = (event.target as HTMLSelectElement).value as OrderStatus; this.orderService.updateStatus(orderId, selectedStatus); } logout(): void { this.auth.logout(); void this.router.navigateByUrl('/login'); } }
